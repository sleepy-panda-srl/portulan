#!/usr/bin/env node
// Acquire the frozen persona comparisons through native children; never run by a recipe.
import fs from "node:fs";
import path from "node:path";
import { createHash, randomUUID } from "node:crypto";
import { spawnSync, execFileSync } from "node:child_process";
import { fileURLToPath, pathToFileURL } from "node:url";
import { cloneTree, childEnv, transcriptOf } from "./warm.mjs";
import { readTranscript } from "./ledger.mjs";
import { CORPUS, grade } from "../evals/ab/personas/grader.mjs";

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
const PARENT_MODEL = "claude-sonnet-5-5";
const ROLES = ["implementer", "reviewer", "librarian"];
const FROZEN = ["evals/ab/personas/corpus.json", "evals/ab/personas/grader.mjs", "cli/persona-routing.test.mjs"];
const digest = bytes => createHash("sha256").update(bytes).digest("hex");
const sha = file => digest(fs.readFileSync(file));
const json = file => JSON.parse(fs.readFileSync(file, "utf8"));
const save = (file, value) => fs.writeFileSync(file, JSON.stringify(value, null, 2) + "\n");
const git = (...args) => execFileSync("git", args, { cwd: ROOT, encoding: "utf8" }).trim();
const records = file => fs.readFileSync(file, "utf8").split("\n").filter(Boolean).map(line => JSON.parse(line));

export function parentPrompt(role, taskFile, effort) {
    const input = {
        subagent_type: `portulan:${role}`, description: "Frozen persona artifact probe",
        prompt: `Read ${taskFile} and follow its task exactly. This is an artifact proposal, not a completed repository change.`,
        run_in_background: false,
        ...(effort === null ? {} : { effort }),
    };
    return "This is a frozen evaluation. Invoke the Agent tool exactly once with these input values: " + JSON.stringify(input) +
        ". Do not supply a child model or change the effort. Wait for that foreground child to finish. Do not answer its task yourself or invoke another child. After completion reply only DONE.";
}

function response(file) {
    const raw = fs.readFileSync(file, "utf8").trim();
    let value;
    try { value = JSON.parse(raw); } catch { throw Error("the CLI returned no single JSON result"); }
    if (value.type !== "result" || value.is_error || value.subtype !== "success" || typeof value.session_id !== "string") throw Error("the CLI did not complete successfully");
    return value;
}

function answerOf(rows) {
    const messages = rows.filter(r => r.type === "assistant" && r.message?.model !== "<synthetic>" && Array.isArray(r.message?.content));
    const last = messages.at(-1)?.message.id;
    if (!last) throw Error("child has no final assistant message");
    const blocks = messages.filter(r => r.message.id === last).flatMap(r => r.message.content).filter(b => b.type === "text");
    return [...new Set(blocks.map(b => b.text))].join("\n").trim();
}

export function nativeInvocation(rows, role, effort) {
    const calls = rows.filter(r => r.type === "assistant").flatMap(r => r.message?.content ?? []).filter(b => b.type === "tool_use" && ["Agent", "Task"].includes(b.name));
    const unique = [...new Map(calls.map(call => [call.id, call])).values()];
    if (unique.length !== 1) throw Error("a parent must invoke exactly one native child");
    const input = unique[0].input;
    if (input?.subagent_type !== `portulan:${role}` || Object.hasOwn(input, "model") || (input.effort ?? null) !== effort) throw Error("the parent did not select the frozen role/inherited model/effort");
    if (input.run_in_background === true) throw Error("a background child is not a completed foreground comparison");
    return { type: input.subagent_type, effort: input.effort ?? null, prompt: input.prompt };
}

function collectCall(entry, dir, tree, env) {
    if (entry.exit !== 0 || entry.timedOut) throw Error("a parent invocation failed or timed out");
    const result = response(path.join(dir, entry.stdout));
    if (result.session_id !== entry.sessionId) throw Error("the result belongs to a different parent session");
    const transcript = transcriptOf(result.session_id, tree, env);
    if (transcript === null) throw Error("the completed parent transcript is unavailable");
    const rows = records(transcript);
    const invocation = nativeInvocation(rows, entry.role, entry.effort);
    if (invocation.prompt !== entry.childPrompt) throw Error("the child did not receive the frozen prompt verbatim");
    const agentDir = path.join(path.dirname(transcript), result.session_id, "subagents");
    const children = fs.readdirSync(agentDir).filter(name => /^agent-.*\.jsonl$/.test(name));
    if (children.length !== 1) throw Error("the parent has other than one child transcript");
    const child = path.join(agentDir, children[0]);
    const childRows = records(child);
    const parentUsage = readTranscript(transcript), childUsage = readTranscript(child);
    if (parentUsage.malformed || childUsage.malformed) throw Error("a transcript has malformed records");
    const parent = parentUsage.requests.filter(r => !r.synthetic && !r.sidechain);
    const requests = childUsage.requests.filter(r => !r.synthetic);
    if (!parent.length || !requests.length) throw Error("parent or child usage is missing");
    const models = [...new Set([...parent, ...requests].map(r => r.model))];
    if (models.length !== 1 || !models[0]) throw Error("inherited parent/child concrete models disagree or are unknown");
    const efforts = [...new Set(requests.map(r => r.effort))];
    if (entry.effort !== null && (efforts.length !== 1 || efforts[0] !== entry.effort)) throw Error("host usage does not establish the requested child effort");
    const target = path.join(dir, "originals", entry.key);
    fs.mkdirSync(target, { recursive: true });
    fs.copyFileSync(transcript, path.join(target, "parent.jsonl"));
    fs.copyFileSync(child, path.join(target, "child.jsonl"));
    return {
        invocation: { type: invocation.type, effort: invocation.effort, promptSha256: digest(invocation.prompt) },
        model: models[0], efforts, parentEfforts: [...new Set(parent.map(r => r.effort))],
        answer: answerOf(childRows), usage: { parent, child: requests },
        raw: { result: sha(path.join(dir, entry.stdout)), parent: sha(path.join(target, "parent.jsonl")), child: sha(path.join(target, "child.jsonl")) },
        transcriptDuplicates: { parent: parentUsage.duplicates, child: childUsage.duplicates },
    };
}

function partialUsage(entry, dir, tree, env) {
    const transcript = transcriptOf(entry.sessionId, tree, env);
    if (transcript === null) return { parent: [], children: [], unavailable: true };
    const target = path.join(dir, "originals", entry.key);
    fs.mkdirSync(target, { recursive: true });
    fs.copyFileSync(transcript, path.join(target, "parent.jsonl"));
    const agentDir = path.join(path.dirname(transcript), entry.sessionId, "subagents");
    const children = fs.existsSync(agentDir) ? fs.readdirSync(agentDir).filter(name => /^agent-.*\.jsonl$/.test(name)).map(name => {
        const file = path.join(agentDir, name);
        fs.copyFileSync(file, path.join(target, name));
        return { usage: readTranscript(file).requests, sha256: sha(file) };
    }) : [];
    return { parent: readTranscript(transcript).requests, parentSha256: sha(transcript), children, unavailable: false };
}

function initialize(dir) {
    if (fs.existsSync(dir)) throw Error("acquisition directory already exists; no automatic retry or resampling");
    fs.mkdirSync(dir, { recursive: true });
    const frozen = Object.fromEntries(FROZEN.map(rel => [rel, sha(path.join(ROOT, rel))]));
    const tree = cloneTree(ROOT, path.join(dir, "tree"));
    const version = execFileSync("claude", ["--version"], { encoding: "utf8" }).trim();
    const journal = { version: 1, phase: "frozen", started: new Date().toISOString(), source: git("rev-parse", "HEAD"), host: version, frozen,
        parentPolicy: { model: PARENT_MODEL, effort: "high", configuredEstimateCeilingUsd: 1, tools: "Agent,Read" }, instrumentSha256: sha(fileURLToPath(import.meta.url)), calls: [] };
    fs.mkdirSync(path.join(dir, "frozen"));
    for (const rel of FROZEN) fs.copyFileSync(path.join(ROOT, rel), path.join(dir, "frozen", path.basename(rel)));
    fs.copyFileSync(fileURLToPath(import.meta.url), path.join(dir, "frozen", "acquire.mjs"));
    fs.mkdirSync(path.join(dir, "tasks"));
    for (const task of CORPUS.cases) save(path.join(dir, "tasks", task.id + ".json"), { role: task.role, task: task.task });
    save(path.join(dir, "journal.json"), journal);
    return { journal, tree };
}

function call(journal, dir, tree, { key, role, effort, task, model = PARENT_MODEL }) {
    const taskFile = path.join(dir, "tasks", task + ".json");
    const env = childEnv(process.env, { cache_lifetime: "5m" }, { local: true });
    const childPrompt = `Read ${taskFile} and follow its task exactly. This is an artifact proposal, not a completed repository change.`;
    const sessionId = randomUUID();
    const args = ["-p", parentPrompt(role, taskFile, effort), "--output-format", "json", "--session-id", sessionId, "--effort", "high", "--plugin-dir", tree,
        "--tools", "Agent,Read", "--allowedTools", `Agent(portulan:${role}),Read`, "--add-dir", path.join(dir, "tasks"), "--max-budget-usd", "1", ...(model ? ["--model", model] : [])];
    const entry = { key, role, effort, task, childPrompt, model, sessionId, argv: args, instrumentSha256: sha(fileURLToPath(import.meta.url)), started: new Date().toISOString(), state: "started", stdout: key + ".stdout.json", stderr: key + ".stderr.txt" };
    journal.calls.push(entry);
    save(path.join(dir, "journal.json"), journal);
    const started = Date.now();
    const run = spawnSync("claude", args, { cwd: tree, env, encoding: "utf8", timeout: 180000, maxBuffer: 4 * 1024 * 1024 });
    fs.writeFileSync(path.join(dir, entry.stdout), run.stdout ?? "");
    fs.writeFileSync(path.join(dir, entry.stderr), run.stderr ?? "");
    Object.assign(entry, { exit: run.status, timedOut: run.error?.code === "ETIMEDOUT", durationMs: Date.now() - started, ended: new Date().toISOString() });
    try {
        entry.collected = collectCall(entry, dir, tree, env);
        entry.state = "complete";
        if (task !== "mechanism") entry.grade = grade(CORPUS.cases.find(c => c.id === task), entry.collected.answer);
    } catch (error) {
        entry.state = "invalid";
        entry.fault = error.message;
        try { entry.partial = partialUsage(entry, dir, tree, env); } catch { entry.partial = { unavailable: true }; }
        journal.phase = "aborted";
    }
    save(path.join(dir, "journal.json"), journal);
    process.stdout.write(`${key}: ${entry.state}${entry.grade ? `, semantic ${entry.grade.ok ? "PASS" : "FAIL"}` : ""}\n`);
    if (entry.state !== "complete") throw Error(entry.fault);
    return entry;
}

export function run(argv = process.argv.slice(2)) {
    const [mode, named] = argv;
    if (!["probe", "run"].includes(mode) || !named || argv.length !== 2) throw Error("usage: node cli/acquire-personas.mjs probe|run <private-directory>");
    const dir = path.resolve(named);
    if (mode === "probe") {
        const { journal, tree } = initialize(dir);
        save(path.join(dir, "tasks", "mechanism.json"), { role: "implementer", task: "Read your role charter, then reply exactly MECHANISM_OK. This tests native effort selection only; do not edit files or call other tools." });
        const entry = call(journal, dir, tree, { key: "mechanism-medium", role: "implementer", effort: "medium", task: "mechanism" });
        if (entry.collected.answer !== "MECHANISM_OK") throw Error("mechanism child did not finish its fixed task");
        journal.model = entry.collected.model;
        journal.phase = "mechanism-demonstrated";
        save(path.join(dir, "journal.json"), journal);
        return 0;
    }
    const journal = json(path.join(dir, "journal.json"));
    if (journal.phase !== "mechanism-demonstrated" || journal.calls.length !== 1) throw Error("no valid untouched mechanism probe; refusing retry or resampling");
    for (const rel of FROZEN) if (sha(path.join(ROOT, rel)) !== journal.frozen[rel]) throw Error("frozen semantic source changed");
    journal.instrument = sha(fileURLToPath(import.meta.url));
    journal.phase = "acquiring";
    save(path.join(dir, "journal.json"), journal);
    const tree = path.join(dir, "tree");
    for (const [i, task] of CORPUS.cases.entries()) {
        for (const arm of i % 2 === 0 ? ["a", "b"] : ["b", "a"]) {
            const entry = call(journal, dir, tree, { key: `${task.id}-${arm}`, role: task.role, effort: arm === "a" ? null : "medium", task: task.id });
            if (entry.collected.model !== journal.model) throw Error("the frozen concrete model changed");
            if (arm === "a" && !entry.grade.ok) { journal.phase = "aborted"; save(path.join(dir, "journal.json"), journal); throw Error("a control failed: no routing is earned and no resampling is allowed"); }
        }
    }
    journal.phase = "complete";
    save(path.join(dir, "journal.json"), journal);
    return 0;
}
if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
    try { process.exitCode = run(); } catch (error) { process.stderr.write(`persona acquisition: ${error.message}\n`); process.exitCode = 2; }
}
